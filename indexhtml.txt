        // Generador de Nieve Navideña (Copos rojos y blancos)
        const snowContainer = document.getElementById('snow-field');
        function createFlake() {
            if (!snowContainer) return;

            flake.classList.add('flake');
            const isBerry = Math.random() > 0.95;
            flake.innerHTML = isBerry ? '🍓' : '❄';
            flake.style.color = isBerry ? '#ff0000' : '#ffffff';
            flake.style.left = Math.random() * 100 + 'vw';
            flake.style.animationDuration = (Math.random() * 5 + 5) + 's';
            flake.style.opacity = Math.random() * 0.4 + 0.1;
            flake.style.fontSize = (Math.random() * 14 + 10) + 'px';
            snowContainer.appendChild(flake);
            setTimeout(() => flake.remove(), 10000);
        }
